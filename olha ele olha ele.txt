create:
// =====================================================
// MOVIMENTO
// =====================================================

up = noone;
down = noone;
left = noone;
right = noone;

velh = 0;
velv = 0;

vel = 1;


// =====================================================
// VIDA
// =====================================================

vida_max = 100;
vida = vida_max;

invencivel = false;
invencivel_timer = 0;


// =====================================================
// MORTE / REVIVER
// =====================================================

morto = false;

reviver_timer = 0;
reviver_tempo = room_speed * 3;

mensagem_morte = "";


// =====================================================
// DASH
// =====================================================

dash_speed = 4;
dash_duration = 10;
dash_recovery = 30;

dash_timer = 0;
dash_recovery_timer = 0;

dash_dir_x = 0;
dash_dir_y = 0;


// =====================================================
// VOO
// =====================================================

voando = false;
voo_vel = 1;


// =====================================================
// ATAQUE
// =====================================================

atacando = false;

ataque_timer = 0;
ataque_duracao = 8;

ataque_dir_x = 0;
ataque_dir_y = 0;

ataque_dano = 10;
ataque_distancia = 20;

step:
// =====================================================
// MORTE / REVIVER
// =====================================================

// Se a vida chegou a zero
if (vida <= 0 && !morto)
{
    vida = 0;

    morto = true;

    // Para tudo
    velh = 0;
    velv = 0;

    atacando = false;
    ataque_timer = 0;

    dash_timer = 0;
    dash_recovery_timer = 0;

    // 3 segundos
    reviver_timer = room_speed * 3;

    mensagem_morte = "VOCE MORREU!";
}


// =====================================================
// SE ESTIVER MORTO
// =====================================================

if (morto)
{
    velh = 0;
    velv = 0;

    reviver_timer--;

    // Reviver
    if (reviver_timer <= 0)
    {
        vida = vida_max;

        morto = false;

        mensagem_morte = "";

        // Invencibilidade por 2 segundos
        invencivel = true;
        invencivel_timer = room_speed * 2;
    }
}


// =====================================================
// INVENCIBILIDADE
// =====================================================

if (invencivel_timer > 0)
{
    invencivel_timer--;
}

if (invencivel_timer <= 0)
{
    invencivel = false;
}


// =====================================================
// SE ESTIVER MORTO, NÃO FAZ MAIS NADA
// =====================================================

if (morto)
{
    exit;
}


// =====================================================
// TECLAS
// =====================================================

var _direita  = keyboard_check(vk_right) || keyboard_check(ord("D"));
var _esquerda = keyboard_check(vk_left)  || keyboard_check(ord("A"));
var _baixo    = keyboard_check(vk_down)  || keyboard_check(ord("S"));
var _cima     = keyboard_check(vk_up)    || keyboard_check(ord("W"));

var _move_x = _direita - _esquerda;
var _move_y = _baixo - _cima;


// =====================================================
// VOO
// =====================================================

if (keyboard_check_pressed(vk_space))
{
    voando = !voando;
}


// =====================================================
// COOLDOWN DO DASH
// =====================================================

if (dash_recovery_timer > 0)
{
    dash_recovery_timer--;
}


// =====================================================
// ATAQUE
// J
// =====================================================

if (keyboard_check_pressed(ord("J")) && !atacando)
{
    atacando = true;
    ataque_timer = ataque_duracao;


    if (_move_x != 0)
    {
        ataque_dir_x = _move_x;
        ataque_dir_y = 0;
    }
    else if (_move_y != 0)
    {
        ataque_dir_x = 0;
        ataque_dir_y = _move_y;
    }
    else
    {
        ataque_dir_x = 1;
        ataque_dir_y = 0;
    }


    // =================================================
    // HITBOX
    // =================================================

    var _ataque_x = x + ataque_dir_x * ataque_distancia;
    var _ataque_y = y + ataque_dir_y * ataque_distancia;


    var _inimigo = instance_place(
        _ataque_x,
        _ataque_y,
        object_enemy1
    );


    if (_inimigo != noone)
    {
        _inimigo.vida -= ataque_dano;
    }
}


// =====================================================
// ATAQUE TIMER
// =====================================================

if (atacando)
{
    ataque_timer--;

    if (ataque_timer <= 0)
    {
        atacando = false;
    }
}


// =====================================================
// DASH
// E
// =====================================================

if (
    keyboard_check_pressed(ord("E"))
    && dash_timer <= 0
    && dash_recovery_timer <= 0
    && !atacando
)
{
    if (_move_x != 0)
    {
        dash_dir_x = _move_x;
        dash_dir_y = 0;
    }
    else if (_move_y != 0)
    {
        dash_dir_x = 0;
        dash_dir_y = _move_y;
    }
    else
    {
        dash_dir_x = 1;
        dash_dir_y = 0;
    }

    dash_timer = dash_duration;
    dash_recovery_timer = dash_recovery;
}


// =====================================================
// DASH
// =====================================================

if (dash_timer > 0)
{
    velh = dash_dir_x * dash_speed;
    velv = dash_dir_y * dash_speed;

    dash_timer--;
}


// =====================================================
// ATAQUE
// =====================================================

else if (atacando)
{
    velh = _move_x;
    velv = _move_y;
}


// =====================================================
// MOVIMENTO NORMAL
// =====================================================

else
{
    var _velocidade = 2;

    velh = 0;
    velv = 0;


    if (voando)
    {
        velh = _move_x * voo_vel;
        velv = _move_y * voo_vel;
    }
    else if (_move_x != 0 || _move_y != 0)
    {
        var _dir = point_direction(
            0,
            0,
            _move_x,
            _move_y
        );

        velh = lengthdir_x(
            _velocidade,
            _dir
        );

        velv = lengthdir_y(
            _velocidade,
            _dir
        );


        image_speed = 1;


        if (_move_x > 0)
        {
            sprite_index = spr_player_walk_right;
        }
        else if (_move_x < 0)
        {
            sprite_index = spr_player_walk_left;
        }
        else if (_move_y > 0)
        {
            sprite_index = spr_player_walk_down;
        }
        else if (_move_y < 0)
        {
            sprite_index = spr_player_walk_up;
        }
    }
    else
    {
        image_speed = 0;
        image_index = 0;
    }
}


// =====================================================
// COLISÃO
// =====================================================

if (voando)
{
    x += velh;
    y += velv;
}
else
{
    move_and_collide(
        velh,
        velv,
        obj_colisor
    );
}


draw:
// =====================================================
// DESENHAR JOGADOR
// =====================================================

if (!morto)
{
    draw_self();
}
else
{
    draw_set_alpha(0.35);

    draw_self();

    draw_set_alpha(1);
}


// =====================================================
// MENSAGEM
// =====================================================

if (morto)
{
    draw_set_halign(fa_center);
    draw_set_valign(fa_middle);

    draw_set_color(c_white);


    draw_text(
        x,
        y - 50,
        "VOCE MORREU!"
    );


    draw_text(
        x,
        y - 25,
        "Reviver em " +
        string(ceil(reviver_timer / room_speed))
    );


    draw_set_halign(fa_left);
    draw_set_valign(fa_top);

    draw_set_color(c_white);
}





create do inimigo:
// =====================================================
// VIDA
// =====================================================

vida_max = 30;
vida = vida_max;


// =====================================================
// MOVIMENTO
// =====================================================

velocidade = 1;


// =====================================================
// ATAQUE
// =====================================================

dano = 50;

distancia_ataque = 30;

ataque_cooldown = 30;
ataque_timer = 0;

step do inimigo:
// =====================================================
// ENCONTRAR JOGADOR
// =====================================================

var _player = instance_nearest(
    x,
    y,
    object_player
);


// =====================================================
// COOLDOWN
// =====================================================

if (ataque_timer > 0)
{
    ataque_timer--;
}


// =====================================================
// JOGADOR EXISTE?
// =====================================================

if (_player != noone)
{
    var _distancia = point_distance(
        x,
        y,
        _player.x,
        _player.y
    );


    // =================================================
    // PERSEGUIR
    // =================================================

    if (!_player.morto && _distancia > distancia_ataque)
    {
        var _direcao = point_direction(
            x,
            y,
            _player.x,
            _player.y
        );


        var _vel_x = lengthdir_x(
            velocidade,
            _direcao
        );


        var _vel_y = lengthdir_y(
            velocidade,
            _direcao
        );


        move_and_collide(
            _vel_x,
            _vel_y,
            obj_colisor
        );
    }


    // =================================================
    // ATAQUE
    // =================================================

    if (
        !_player.morto
        && _distancia <= distancia_ataque
        && ataque_timer <= 0
    )
    {
        // Só causa dano se não estiver invencível
        if (!_player.invencivel)
        {
            _player.vida -= dano;

            // Segurança
            if (_player.vida < 0)
            {
                _player.vida = 0;
            }

            // Invencibilidade
            _player.invencivel = true;

            _player.invencivel_timer = 30;

            // Cooldown
            ataque_timer = ataque_cooldown;
        }
    }
}


// =====================================================
// MORTE DO INIMIGO
// =====================================================

if (vida <= 0)
{
    instance_destroy();
}
