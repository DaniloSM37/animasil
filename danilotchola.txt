create:up = noone;
down = noone;
left = noone;
right = noone;

velh = 0;
velv = 0;

vel = 1;


// =====================================================
// DASH
// =====================================================

dash_speed = 6;
dash_duration = 8;

// Tempo de recuperação depois do dash
dash_recovery = 30;

dash_timer = 0;
dash_recovery_timer = 0;

dash_dir_x = 0;
dash_dir_y = 0;


// =====================================================
// ANIMAÇÃO DO DASH
// =====================================================

// Coloque aqui o sprite da animação
dash_sprite_right = spr_player_dash_right;
dash_sprite_left  = spr_player_dash_left;

dash_sprite_down  = spr_player_dash_down;





step:// =====================================================
// TECLAS
// =====================================================

var _direita  = keyboard_check(vk_right) || keyboard_check(ord("D"));
var _esquerda = keyboard_check(vk_left)  || keyboard_check(ord("A"));
var _baixo    = keyboard_check(vk_down)  || keyboard_check(ord("S"));
var _cima     = keyboard_check(vk_up)    || keyboard_check(ord("W"));

var _move_x = _direita - _esquerda;
var _move_y = _baixo - _cima;


// =====================================================
// RECUPERAÇÃO DO DASH
// =====================================================

if (dash_recovery_timer > 0)
{
    dash_recovery_timer--;
}


// =====================================================
// COMEÇAR O DASH
// =====================================================

if (keyboard_check_pressed(ord("E"))
&& dash_timer <= 0
&& dash_recovery_timer <= 0)
{
    // =================================================
    // ESCOLHE UMA ÚNICA DIREÇÃO
    // =================================================

    // Prioridade horizontal
    if (_move_x != 0)
    {
        dash_dir_x = _move_x;
        dash_dir_y = 0;

        // Animação
        if (_move_x > 0)
            sprite_index = dash_sprite_right;
        else
            sprite_index = dash_sprite_left;
    }
    
    // Depois vertical
    else if (_move_y != 0)
    {
        dash_dir_x = 0;
        dash_dir_y = _move_y;

        // Animação
        if (_move_y < 0)
            sprite_index = dash_sprite_up;
        else
            sprite_index = dash_sprite_down;
    }
    
    // Se não apertar nenhuma direção
    else
    {
        // Dash para a direita
        dash_dir_x = 1;
        dash_dir_y = 0;

        sprite_index = dash_sprite_right;
    }


    // Começa o dash
    dash_timer = dash_duration;

    // Começa a recuperação
    dash_recovery_timer = dash_recovery;

    // Começa a animação
    image_index = 0;
    image_speed = 1;
}


// =====================================================
// DURANTE O DASH
// =====================================================

if (dash_timer > 0)
{
    velh = dash_dir_x * dash_speed;
    velv = dash_dir_y * dash_speed;

    dash_timer--;

    image_speed = 1;
}


// =====================================================
// MOVIMENTO NORMAL
// =====================================================

else
{
    var _velocidade = 2;

    velh = 0;
    velv = 0;


    if (_move_x != 0 || _move_y != 0)
    {
        // Movimento normal continua podendo ser diagonal
        var _dir = point_direction(0, 0, _move_x, _move_y);

        velh = lengthdir_x(_velocidade, _dir);
        velv = lengthdir_y(_velocidade, _dir);


        // =================================================
        // SPRITES NORMAIS
        // =================================================

        image_speed = 1;

        if (_move_x > 0)
            sprite_index = spr_player_walk_right;

        else if (_move_x < 0)
            sprite_index = spr_player_walk_left;

        else if (_move_y > 0)
            sprite_index = spr_player_walk_down;

        else if (_move_y < 0)
            sprite_index = spr_player_walk_up;
    }
    else
    {
        image_speed = 0;
        image_index = 0;
    }
}


// =====================================================
// MOVIMENTO E COLISÃO
// =====================================================

move_and_collide(velh, velv, obj_colisor);
